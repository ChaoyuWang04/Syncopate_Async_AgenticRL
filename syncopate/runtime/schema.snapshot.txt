FUNCTION notify_run_event
  CREATE OR REPLACE FUNCTION public.notify_run_event()
   RETURNS trigger
   LANGUAGE plpgsql
  AS $function$
  BEGIN
    PERFORM pg_notify('run_events', NEW.org_id || '|' || NEW.run_id);
    RETURN NULL;
  END; $function$
FUNCTION touch_updated_at
  CREATE OR REPLACE FUNCTION public.touch_updated_at()
   RETURNS trigger
   LANGUAGE plpgsql
  AS $function$
  BEGIN
    NEW.updated_at = now();
    RETURN NEW;
  END; $function$
